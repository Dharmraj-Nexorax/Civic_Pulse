import { NextRequest } from 'next/server';
import { z } from 'zod';
import { query } from '@/lib/db';
import { config } from '@/lib/config';
import { verifyPassword, createAuthToken, normalizeMobile, AuthUser } from '@/lib/auth';
import { okResponse, errorResponse } from '@/lib/http';
import { unauthenticated } from '@/lib/errors';

export const dynamic = 'force-dynamic';

const LoginSchema = z.object({
  identifier: z.string().min(1, 'Identifier is required'),
  password: z.string().min(1, 'Password is required'),
  remember: z.boolean().optional(),
});

export async function POST(req: NextRequest) {
  try {
    const body = await req.json();
    const data = LoginSchema.parse(body);

    const normMobile = normalizeMobile(data.identifier);
    const normEmail = data.identifier.trim().toLowerCase();

    // Query user by mobile or email
    const { rows: users } = await query<AuthUser & { password_hash: string }>(
      `SELECT u.id, u.organization_id, u.role, u.full_name, u.mobile, u.email,
              u.ward_id, u.team_id, u.password_hash, u.is_active
       FROM users u
       WHERE (u.mobile = $1 OR u.email = $2) AND u.is_active = true`,
      [normMobile, normEmail]
    );

    if (users.length === 0) {
      throw unauthenticated('Invalid credentials.');
    }

    const user = users[0];
    const passwordValid = await verifyPassword(data.password, user.password_hash);
    if (!passwordValid) {
      throw unauthenticated('Invalid credentials.');
    }

    const authUser: AuthUser = {
      id: user.id,
      organization_id: user.organization_id,
      role: user.role,
      full_name: user.full_name,
      mobile: user.mobile,
      email: user.email,
      ward_id: user.ward_id,
      team_id: user.team_id,
    };

    const token = await createAuthToken(authUser, !!data.remember);

    const res = okResponse({ user: authUser });
    const cookieOptions: any = {
      name: config.cookieName,
      value: token,
      httpOnly: true,
      sameSite: 'lax',
      path: '/',
      secure: process.env.NODE_ENV === 'production',
    };

    if (data.remember) {
      cookieOptions.maxAge = config.cookieMaxAgeSeconds;
    }

    res.cookies.set(cookieOptions);
    return res;
  } catch (error) {
    return errorResponse(error);
  }
}

    const user = await requireRole(req, ['CITIZEN']);
    const body = await req.json();
    const { description } = AssistSchema.parse(body);

    const { rows: categories } = await query<{
      id: string;
      key: string;
      name: string;
      keywords: string[];
    }>(
      'SELECT id, key, name, keywords FROM categories WHERE organization_id = $1 AND is_active = true',
      [user.organization_id]
    );

    // Try AI service first
    const aiResult = await aiCategorize(
      description,
      categories.map((c) => ({ key: c.key, name: c.name, keywords: c.keywords }))
    );

    if (aiResult && aiResult.ai_available) {
      const matchedCat = categories.find((c) => c.key === aiResult.category_key);
      return okResponse({
        ai_available: true,
        method: aiResult.method,
        category_key: aiResult.category_key,
        category_id: matchedCat ? matchedCat.id : null,

        cleaned_description: aiResult.cleaned_description ?? description.trim(),
        missing_info: aiResult.missing_info ?? [],
      });
    }

    // Deterministic fallback: keyword categorizer + template
    const kwResult = categorizeByKeywords(description, categories);
    const matchedCat = categories.find((c) => c.id === kwResult.categoryId);
    const tmpl = generateFallbackAssist(description, matchedCat?.name, null);

    return okResponse({
      ai_available: false,
      method: 'keyword',
      category_key: kwResult.categoryKey,
      category_id: kwResult.categoryId,
      confidence: null,

export async function GET(req: NextRequest) {
  try {
    const user = await requireRole(req, ['CITIZEN']);
    const { searchParams } = new URL(req.url);
    let lat = parseFloat(searchParams.get('lat') || '');
    let lng = parseFloat(searchParams.get('lng') || '');
    const radiusMeters = parseFloat(searchParams.get('radius') || '5000');

    if (isNaN(lat) || isNaN(lng) || (lat === 0 && lng === 0)) {
      lat = MANDSAUR_CONFIG.centerLat;
      lng = MANDSAUR_CONFIG.centerLng;
    }

    const { rows } = await query(
      `SELECT i.id as issue_id, i.lat, i.lng, i.status, cat.name as category_name, w.name as ward_name
       FROM issues i
       JOIN categories cat ON cat.id = i.category_id
       JOIN wards w ON w.id = i.ward_id
       WHERE i.organization_id = $1 AND i.status <> 'CITIZEN_VERIFIED' AND i.merged_into_issue_id IS NULL`,
      [user.organization_id]
    );

    const issues = rows
      .map(r => {

          issue_id: r.issue_id,
          category_name: r.category_name,
          ward_name: r.ward_name,
          status: r.status,
          status_label: r.status.replace(/_/g, ' '),
          distance_m: dist,
        };
      })
      .filter(r => r.distance_m <= radiusMeters)
      .sort((a, b) => a.distance_m - b.distance_m);

    return okResponse({ issues });

    }
    const orgId = orgRows[0].id;

    const days = isNaN(daysParam) ? 90 : daysParam;

                         AND i.organization_id = $1
                         AND i.merged_into_issue_id IS NULL
                         AND i.created_at >= now() - ($2 || ' days')::interval
       LEFT JOIN categories cat ON cat.id = i.category_id ${categoryFilter}
       LEFT JOIN complaints c ON c.issue_id = i.id
       WHERE w.organization_id = $1

import { requireRole } from '@/lib/auth';
import { withTransaction } from '@/lib/db';
import { startWork } from '@/services/resolutionService';
import { errorResponse, okResponse } from '@/lib/http';

export const dynamic = 'force-dynamic';

export async function POST(
  req: NextRequest,
  { params }: { params: { id: string } }
) {
  try {
    const user = await requireRole(req, ['WORKER']);
    await withTransaction(async (client) => {
      await startWork(client, params.id, user.id);
    });

    return okResponse({ message: 'Work started successfully' });
  } catch (error) {
    return errorResponse(error);
  }
}
